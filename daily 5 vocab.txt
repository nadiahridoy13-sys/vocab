<script>
  // ===== WORD DATA =====
  const allWords = [
    {
      word: "meticulous",
      phonetic: "/məˈtɪkjələs/",
      meaning: "showing great attention to detail; very careful and precise",
      synonyms: ["careful", "thorough", "precise"],
      example: "The researcher was meticulous in recording the data."
    },
    {
      word: "ubiquitous",
      phonetic: "/juːˈbɪkwɪtəs/",
      meaning: "present, appearing, or found everywhere",
      synonyms: ["everywhere", "omnipresent", "pervasive"],
      example: "Smartphones have become ubiquitous in modern society."
    },
    {
      word: "alleviate",
      phonetic: "/əˈliːvieɪt/",
      meaning: "to make suffering, problems, or pain less severe",
      synonyms: ["ease", "reduce", "relieve"],
      example: "The new policy aims to alleviate traffic congestion."
    },
    {
      word: "profound",
      phonetic: "/prəˈfaʊnd/",
      meaning: "very great or intense; having deep meaning",
      synonyms: ["deep", "intense", "significant"],
      example: "The discovery had a profound effect on science."
    },
    {
      word: "deteriorate",
      phonetic: "/dɪˈtɪəriəreɪt/",
      meaning: "to become worse over time",
      synonyms: ["worsen", "decline", "degenerate"],
      example: "Air quality continues to deteriorate in big cities."
    },
    {
      word: "compulsory",
      phonetic: "/kəmˈpʌlsəri/",
      meaning: "required by law or rule; mandatory",
      synonyms: ["mandatory", "obligatory", "required"],
      example: "Wearing a helmet is compulsory for motorcyclists."
    },
    {
      word: "fluctuate",
      phonetic: "/ˈflʌktʃueɪt/",
      meaning: "to change or vary frequently",
      synonyms: ["vary", "shift", "oscillate"],
      example: "Oil prices tend to fluctuate throughout the year."
    },
    {
      word: "inevitable",
      phonetic: "/ɪnˈevɪtəbl/",
      meaning: "certain to happen; unavoidable",
      synonyms: ["unavoidable", "certain", "inescapable"],
      example: "Change is inevitable in every industry."
    },
    {
      word: "substantial",
      phonetic: "/səbˈstænʃl/",
      meaning: "of considerable importance, size, or worth",
      synonyms: ["significant", "considerable", "large"],
      example: "There has been a substantial increase in demand."
    },
    {
      word: "vulnerable",
      phonetic: "/ˈvʌlnərəbl/",
      meaning: "exposed to the possibility of being harmed",
      synonyms: ["at risk", "defenseless", "exposed"],
      example: "Children are particularly vulnerable to online risks."
    }
  ];

  // Select 5 words based on day (simple daily rotation)
  function getDailyWords() {
    const day = Math.floor(Date.now() / 86400000); // days since 1970
    const start = day % allWords.length;
    const selected = [];
    for (let i = 0; i < 5; i++) {
      selected.push(allWords[(start + i) % allWords.length]);
    }
    return selected;
  }

  const dailyWords = getDailyWords();
  let questions = [];
  let currentQ = 0;
  let score = 0;
  let wrongWords = [];
  let answered = false;

  // ===== HOME =====
  function renderHome() {
    const container = document.getElementById("words-container");
    container.innerHTML = "";
    dailyWords.forEach(w => {
      container.innerHTML += `
        <div class="card">
          <div class="word-row">
            <div>
              <div class="word">${w.word}</div>
              <div class="phonetic">${w.phonetic}</div>
            </div>
            <button class="speak-btn" onclick="speak('${w.word}')">🔊</button>
          </div>
          <div class="label">Meaning</div>
          <div class="meaning">${w.meaning}</div>
          <div class="label">Synonyms</div>
          <div class="synonyms">${w.synonyms.join(", ")}</div>
          <div class="label">Example</div>
          <div class="example">${w.example}</div>
        </div>
      `;
    });
  }

  function speak(text) {
    if ("speechSynthesis" in window) {
      const u = new SpeechSynthesisUtterance(text);
      u.lang = "en-US";
      u.rate = 0.85;
      speechSynthesis.speak(u);
    }
  }

  // ===== QUIZ =====
  function startQuiz() {
    questions = [];
    dailyWords.forEach(w => {
      // Meaning question
      let opts = [w.meaning];
      const others = dailyWords.filter(x => x.word !== w.word).map(x => x.meaning);
      opts = opts.concat(others.sort(() => 0.5 - Math.random()).slice(0, 3));
      opts.sort(() => 0.5 - Math.random());
      questions.push({
        text: What is the meaning of "${w.word}"?,
        options: opts,
        answer: w.meaning,
        word: w
      });

      // Synonym question
      if (w.synonyms.length) {
        const correct = w.synonyms[0];
        let sopts = [correct];
        const otherSyn = dailyWords
          .filter(x => x.word !== w.word)
          .map(x => x.synonyms[0] || x.word);
        sopts = sopts.concat(otherSyn.sort(() => 0.5 - Math.random()).slice(0, 3));
        sopts.sort(() => 0.5 - Math.random());
        questions.push({
          text: Which is a synonym of "${w.word}"?,
          options: sopts,
          answer: correct,
          word: w
        });
      }
    });

    questions.sort(() => 0.5 - Math.random());
    currentQ = 0;
    score = 0;
    wrongWords = [];
    answered = false;

    document.getElementById("home-screen").style.display = "none";
    document.getElementById("result-screen").style.display = "none";
    document.getElementById("quiz-screen").style.display = "block";
    showQuestion();
  }

  function showQuestion() {
    answered = false;
    const q = questions[currentQ];
    document.getElementById("question-text").innerText = q.text;
    document.getElementById("progress-bar").style.width = ((currentQ / questions.length) * 100) + "%";
    document.getElementById("next-btn").style.display = "none";

    const container = document.getElementById("options-container");
    container.innerHTML = "";
    q.options.forEach(opt => {
      const div = document.createElement("div");
      div.className = "option";
      div.innerText = opt;
      div.onclick = () => selectAnswer(div, opt, q);
      container.appendChild(div);
    });
  }

  function selectAnswer(div, selected, q) {
    if (answered) return;
    answered = true;

    const options = document.querySelectorAll(".option");
    options.forEach(o => {
      if (o.innerText === q.answer) o.classList.add("correct");
      if (o.innerText === selected && selected !== q.answer) o.classList.add("wrong");
    });

    if (selected === q.answer) {
      score++;
    } else {
      if (!wrongWords.find(w => w.word === q.word.word)) {
        wrongWords.push(q.word);
      }
    }

    document.getElementById("next-btn").style.display = "block";
    document.getElementById("next-btn").innerText =
      currentQ + 1 >= questions.length ? "See Result" : "Next";
  }

  function nextQuestion() {
    currentQ++;
    if (currentQ >= questions.length) {
      showResult();
    } else {
      showQuestion();
    }
  }

  function showResult() {
    document.getElementById("quiz-screen").style.display = "none";
    document.getElementById("result-screen").style.display = "block";

    const percent = Math.round((score / questions.length) * 100);
    document.getElementById("score-percent").innerText = percent + "%";
    document.getElementById("score-percent").style.color = percent >= 70 ? "#34c759" : "#ff9500";
    document.getElementById("score-text").innerText = ${score} / ${questions.length} correct;

    if (wrongWords.length) {
      document.getElementById("review-box").style.display = "block";
      document.getElementById("wrong-list").innerHTML = wrongWords
        .map(w => <div>• ${w.word}</div>)
        .join("");
    } else {
      document.getElementById("review-box").style.display = "none";
    }
  }

  function goHome() {
    document.getElementById("result-screen").style.display = "none";
    document.getElementById("home-screen").style.display = "block";
  }

  // Start
  renderHome();
</script>
